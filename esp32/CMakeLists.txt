cmake_minimum_required(VERSION 4.0.0)

execute_process(
    COMMAND git rev-list --count HEAD
    WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/.."
    OUTPUT_VARIABLE SWIRSKI_BUILD_NUMBER
    OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE SWIRSKI_BUILD_NUMBER_RESULT
)

execute_process(
    COMMAND git rev-parse --short=7 HEAD
    WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/.."
    OUTPUT_VARIABLE SWIRSKI_BUILD_SHA
    OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE SWIRSKI_BUILD_SHA_RESULT
)

if(
    SWIRSKI_BUILD_NUMBER_RESULT EQUAL 0 AND
    SWIRSKI_BUILD_SHA_RESULT EQUAL 0
)
    set(PROJECT_VER "b${SWIRSKI_BUILD_NUMBER}-${SWIRSKI_BUILD_SHA}")
else()
    set(PROJECT_VER "b0-unknown")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

project(swirski_os_esp32)
